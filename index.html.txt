<script>

function addRow(){
let table = document.getElementById("billTable");

let row = `
<tr>
<td><input type="text" class="form-control"></td>
<td><input type="number" class="form-control price" oninput="calculate()"></td>
<td><input type="number" class="form-control qty" oninput="calculate()"></td>
<td><input type="number" class="form-control discount" oninput="calculate()"></td>
<td class="subtotal">0</td>
</tr>
`;

table.innerHTML += row;
}

function calculate(){
let prices = document.querySelectorAll(".price");
let qtys = document.querySelectorAll(".qty");
let discounts = document.querySelectorAll(".discount");
let subtotals = document.querySelectorAll(".subtotal");

let total = 0;

for(let i=0; i<prices.length; i++){

let price = prices[i].value || 0;
let qty = qtys[i].value || 0;
let discount = discounts[i].value || 0;

let subtotal = (price * qty) - discount;
subtotals[i].innerText = subtotal;

total += subtotal;
}

document.getElementById("total").innerText = total;
}
